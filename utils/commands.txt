public Sub riempi_react
	Private a As Char
	Private stringa_scrittura As String
	riempi_batt
	a = Mid(Main.messaggio_ble,4,1)
	If a = "E" Then
		MsgboxAsync ("NON SENTO ATLETA","MISURA")
	End If
	If a = "O" Then
		Main.t_m = 0
		clv1.clear
		clv1.AddTextItem("MISURA","value 1")
	End If
	
	If a = "U" Then 'fine misura
		Button_discard.enabled = True
		dati_presenti = 1
	End If


	If a = "M" Then
		dati_presenti = 1
		Main.t_volo = Mid(Main.messaggio_ble,5,4) 
		Main.t_contatto = Mid(Main.messaggio_ble,9,4)
		'aggiorno per il server
		Main.t_m = Main.t_m + 1
		Log ("t_m " & Main.t_m)
		Main.t_v(Main.t_m) = Main.t_volo
		Main.t_c(Main.t_m) = Main.t_contatto
		'aggiunta calcolo potenza
		Main.media_volo = Main.media_volo + Main.t_volo / 1000
		Main.media_contatto = Main.media_contatto + Main.t_contatto / 1000
		Main.n_salti = Main.n_salti + 1
		If Main.n_salti > 0 Then
			label6.text = Round2((96.2361 * Main.media_volo * (Main.media_volo + Main.media_contatto)) / (4 * Main.n_salti * Main.media_contatto),2)
		Else
			label6.text = ""
		End If
		B4XPages.MainPage.calcola_altezza
		stringa_scrittura = ""
		For i = 1 To (10 - stringa_scrittura.Length)
			stringa_scrittura = stringa_scrittura & " "
		Next
		stringa_scrittura = stringa_scrittura & Main.altezza & " cM      " & Main.t_contatto & " mS"
		clv1.AddTextItem( stringa_scrittura ,"value 1")
	End If
End Sub
